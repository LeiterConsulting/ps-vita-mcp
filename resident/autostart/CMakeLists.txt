cmake_minimum_required(VERSION 3.16)
set(CMAKE_TOOLCHAIN_FILE "$ENV{VITASDK}/share/vita.toolchain.cmake" CACHE PATH "Vita toolchain")
project(vita_autostart C)
include("$ENV{VITASDK}/share/vita.cmake" REQUIRED)
foreach(name CONTROL_BUILD_ID CONTROL_KERNEL_SHA256 CONTROL_SERVICE_SHA256)
  string(LENGTH "${${name}}" length)
  if(NOT length EQUAL 64 OR NOT "${${name}}" MATCHES "^[0-9a-f]+$")
    message(FATAL_ERROR "${name} must be an exact lowercase SHA-256 identity")
  endif()
endforeach()
if(NOT CONTROL_MODULE_ROOT MATCHES "^ux0:data/vita-autostart/setup-[0-9a-f-]+/$")
  message(FATAL_ERROR "CONTROL_MODULE_ROOT must be a fixed immutable autostart setup directory with trailing slash")
endif()
string(LENGTH "${CONTROL_MODULE_ROOT}" root_length)
if(root_length GREATER 120)
  message(FATAL_ERROR "CONTROL_MODULE_ROOT is too long")
endif()
if(NOT EXISTS "${CONTROL_STUB_DIR}/libvita_control_kernel_stub.a")
  message(FATAL_ERROR "CONTROL_STUB_DIR must contain the matched generated Control stubs")
endif()
function(autostart_self target source)
  vita_create_self(${target} ${source} ${ARGN})
endfunction()
add_executable(autostart_shell_probe probe.c core.c)
target_compile_definitions(autostart_shell_probe PRIVATE CONTROL_BUILD_ID="${CONTROL_BUILD_ID}")
target_compile_options(autostart_shell_probe PRIVATE -std=c11 -Wall -Wextra -Werror -O2 -g -fno-builtin -fno-pic -fno-PIC)
set_target_properties(autostart_shell_probe PROPERTIES LINK_FLAGS "-nostdlib -Wl,-q")
target_link_libraries(autostart_shell_probe gcc SceLibKernel_stub_weak SceKernelThreadMgr_stub_weak SceIofilemgr_stub_weak "${CONTROL_STUB_DIR}/libvita_control_kernel_stub.a")
autostart_self(autostart_shell_probe.suprx autostart_shell_probe UNSAFE CONFIG "${CMAKE_CURRENT_SOURCE_DIR}/probe.yml")
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/probe_hash.h"
  COMMAND "${CMAKE_COMMAND}" "-DPROBE=${CMAKE_CURRENT_BINARY_DIR}/autostart_shell_probe.suprx"
          "-DOUTPUT=${CMAKE_CURRENT_BINARY_DIR}/probe_hash.h" -P "${CMAKE_CURRENT_SOURCE_DIR}/write_probe_hash.cmake"
  DEPENDS autostart_shell_probe.suprx "${CMAKE_CURRENT_SOURCE_DIR}/write_probe_hash.cmake")
add_executable(vita_autostart platform_vita.c core.c ../sha256.c "${CMAKE_CURRENT_BINARY_DIR}/probe_hash.h")
target_include_directories(vita_autostart PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(vita_autostart PRIVATE CONTROL_BUILD_ID="${CONTROL_BUILD_ID}" CONTROL_MODULE_ROOT="${CONTROL_MODULE_ROOT}"
  CONTROL_KERNEL_SHA256="${CONTROL_KERNEL_SHA256}" CONTROL_SERVICE_SHA256="${CONTROL_SERVICE_SHA256}")
target_compile_options(vita_autostart PRIVATE -std=c11 -Wall -Wextra -Werror -O2 -g -fno-builtin -fno-pic -fno-PIC)
set_target_properties(vita_autostart PROPERTIES LINK_FLAGS "-nostdlib -Wl,-q")
target_link_libraries(vita_autostart gcc taihen_stub SceLibKernel_stub_weak SceKernelThreadMgr_stub_weak SceIofilemgr_stub_weak SceProcessmgr_stub_weak SceAppMgr_stub)
autostart_self(vita_autostart.suprx vita_autostart UNSAFE CONFIG "${CMAKE_CURRENT_SOURCE_DIR}/plugin.yml")
