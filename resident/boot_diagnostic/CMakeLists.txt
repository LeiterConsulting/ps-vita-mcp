cmake_minimum_required(VERSION 3.16)
set(CMAKE_TOOLCHAIN_FILE "$ENV{VITASDK}/share/vita.toolchain.cmake" CACHE PATH "Vita toolchain")
project(vita_boot_diagnostic C)
include("$ENV{VITASDK}/share/vita.cmake" REQUIRED)
string(LENGTH "${DIAGNOSTIC_BUILD_ID}" id_length)
if(NOT id_length EQUAL 64 OR NOT DIAGNOSTIC_BUILD_ID MATCHES "^[0-9a-f]+$")
  message(FATAL_ERROR "DIAGNOSTIC_BUILD_ID must be an exact lowercase source SHA-256")
endif()
string(LENGTH "${DIAGNOSTIC_MODULE_ROOT}" root_length)
if(root_length GREATER 120 OR NOT DIAGNOSTIC_MODULE_ROOT MATCHES "^ux0:data/vita-boot-diagnostic/setup-[0-9a-f-]+/$")
  message(FATAL_ERROR "DIAGNOSTIC_MODULE_ROOT must be a fixed immutable setup directory with trailing slash")
endif()
add_executable(vita_boot_diagnostic platform_vita.c)
target_compile_definitions(vita_boot_diagnostic PRIVATE DIAGNOSTIC_BUILD_ID="${DIAGNOSTIC_BUILD_ID}" DIAGNOSTIC_MODULE_ROOT="${DIAGNOSTIC_MODULE_ROOT}")
target_compile_options(vita_boot_diagnostic PRIVATE -std=c11 -Wall -Wextra -Werror -O2 -g -fno-builtin -fno-pic -fno-PIC)
set_target_properties(vita_boot_diagnostic PROPERTIES LINK_FLAGS "-nostdlib -Wl,-q")
# Match Control's weak SceLibKernel wrapper binding for mutex/thread calls.
target_link_libraries(vita_boot_diagnostic gcc SceLibKernel_stub_weak SceKernelThreadMgr_stub_weak SceIofilemgr_stub_weak SceProcessmgr_stub_weak SceAppMgr_stub)
vita_create_self(vita_boot_diagnostic.suprx vita_boot_diagnostic UNSAFE CONFIG "${CMAKE_CURRENT_SOURCE_DIR}/plugin.yml")
